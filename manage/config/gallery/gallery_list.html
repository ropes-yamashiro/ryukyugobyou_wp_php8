    列データ
    {thumb_width} = サムネイルの最大横幅
    {thumb_height} = サムネイルの最大縦幅
    ギャラリーのデータ
    {img_mini_path} = サムネイルのイメージのパス
    {img_big_path} = イメージのパス
    {img_width} = イメージの横幅
    {img_height} =　イメージの縦幅
    {img_title} =　イメージのタイトル
    {img_comment1} = イメージのコメント1
    {img_comment2} = イメージのコメント2
    {img_link} = リンク先URL

    その他のデータ
    {today_year} =　今日の年
    {today_mon} =　今日の月
    {today_mday} = 今日の日
    {today_week} = 曜日

    {final_year} =　最終更新日の年
    {final_mon} =　最終更新日の月
    {final_mday} = 最終更新日の日
    {final_week} = 最終更新日の曜日

    {view_id} = 表示ID
    {view_title} = 表示位置のタイトル
    {view_comment} = 表示位置のコメント

    {popup_width} = ポップアップウインドウの横幅
    {popup_height} = ポップアップウインドウの縦幅
    {popup_url} = ポップアップのURL
    {popup_id} =　表示する内容のID
